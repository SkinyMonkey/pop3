   use std::path::{Path, PathBuf};

   /// Generate script filename for a level and tribe.
   /// Format: level_XX_tribe_Y.lua (where XX is zero-padded level, Y is tribe index 0-3)
   pub fn script_filename(level: u32, tribe_index: u8) -> String {
       format!("level_{:02}_tribe_{}.lua", level, tribe_index)
   }

   /// Load a Lua script file from disk.
   pub fn load_tribe_script(path: &Path) -> Result<String, ScriptLoadError> {
       std::fs::read_to_string(path)
           .map_err(|e| ScriptLoadError::Io { path: path.to_path_buf(), source: e.to_string() })
   }

   /// Find all tribe scripts for a given level in the scripts directory.
   /// Returns Vec of (tribe_index, file_path) pairs.
   pub fn find_scripts_for_level(level: u32, scripts_dir: &Path) -> Vec<(u8, PathBuf)> {
       let mut found = Vec::new();
       for tribe in 0..4u8 {
           let filename = script_filename(level, tribe);
           let path = scripts_dir.join(&filename);
           if path.exists() {
               found.push((tribe, path));
           }
       }
       found
   }

   #[derive(Debug)]
   pub enum ScriptLoadError {
       Io { path: PathBuf, source: String },
   }

   impl std::fmt::Display for ScriptLoadError {
       fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
           match self {
               Self::Io { path, source } => write!(f, "Failed to load script {}: {}", path.display(), source),
           }
       }
   }
   impl std::error::Error for ScriptLoadError {}
   ```

   /// Load scripts for a level. Called when transitioning to InGame.
   pub fn load_level_scripts(&mut self, level: u32, scripts_dir: &std::path::Path, player_tribe: u8) {
       self.player_tribe = player_tribe;
       let scripts = crate::data::scripts::find_scripts_for_level(level, scripts_dir);
       for (tribe, path) in scripts {
           if tribe as u8 == player_tribe {
               continue; // Skip player tribe

           match crate::data::scripts::load_tribe_script(&path) {

   impl AiTick for AiSystem {
       /// Mirrors AI_UpdateAllTribes @ 0x0041a7d0
       fn tick_update_ai(&mut self) {
           for tribe_idx in 0..4u8 {
               if tribe_idx == self.player_tribe {
                   continue;
               }
               if !self.tribe_states[tribe_idx as usize].active {
                   continue;
               }
               if !self.scripts_loaded[tribe_idx as usize] {
                   continue;
               }

               // Set current tribe in bridge for PopScript functions
               if let Ok(globals) = self.lua.globals().get::<mlua::Table>("_bridge") {
                   let _ = globals.set("current_tribe", tribe_idx);
               }

               // Execute the tribe's script tick function if defined
               let func_name = format!("_tribe_{}_tick", tribe_idx);
               match self.lua.globals().get::<mlua::Function>(&func_name) {
                   Ok(func) => {
                       if let Err(e) = func.call::<()>(()) {
                           log::error!("AI script error for tribe {}: {}", tribe_idx, e);
                       }
                   }
                   Err(_) => {
                       // No per-tick function; script runs on load only (EVERY handles timing)
                   }
               }
           }
       }
   }
   ```
