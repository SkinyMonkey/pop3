if let Some(ref ai) = self.engine.ai_system {

Replace with:
```rust
if let Some(ref mut ai) = self.engine.ai_system {
    let cmds = ai.drain_pending_commands();
    let player_tribe = self.engine.game_world.player_tribe;
    // Determine which tribe(s) the AI controls. For now, dispatch
    // for all non-player tribes that have active scripts.
    for tribe_idx in 0..4u8 {
        if tribe_idx == player_tribe { continue; }
        if !ai.tribe_states[tribe_idx as usize].active { continue; }
        crate::engine::ai::dispatch::dispatch_ai_commands(
            &cmds,
            &mut self.engine.unit_coordinator,
            ai,
            tribe_idx,
        );
    }
}
```

Note: This requires changing `ref ai` to `ref mut ai` since dispatch_ai_commands needs `&mut AiSystem` (for building_placement and shaman_commands). Also, `unit_coordinator` is accessed via `self.engine.unit_coordinator` — check borrow rules. If there's a conflict because `ai_system` is borrowed from `self.engine` while also borrowing `self.engine.unit_coordinator`, restructure: take `ai_system` out with `take()` temporarily:
```rust
if let Some(mut ai) = self.engine.ai_system.take() {
    let cmds = ai.drain_pending_commands();
    let player_tribe = self.engine.game_world.player_tribe;
    for tribe_idx in 0..4u8 {
        if tribe_idx == player_tribe { continue; }
        if !ai.tribe_states[tribe_idx as usize].active { continue; }
        crate::engine::ai::dispatch::dispatch_ai_commands(
            &cmds,
            &mut self.engine.unit_coordinator,
            &mut ai,
            tribe_idx,
        );
    }
    self.engine.ai_system = Some(ai);
}
```

This is the same temporary-ownership pattern used elsewhere in app.rs for borrow checker conflicts (see DeferredAction and BuildingTickActions patterns).
