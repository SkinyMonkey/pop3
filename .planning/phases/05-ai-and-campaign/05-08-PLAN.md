   ```rust
   // Before simulation tick:
   if let Some(ref mut ai) = self.ai_system {
       ai.update_bridge(/* pass game_tick, tribe_populations, tribe_mana, etc. from GameWorld */);
   }

   // Then in TickSubsystems:

       Some(sys) => sys,
       None => &mut NoOp,
   };
   // Pass ai to TickSubsystems
   ```
   The update_bridge call populates the AiGameBridge with current game state (tick counter, tribe populations, mana, etc.) so that PopScript functions read fresh values during script execution.

4. Wire save/load GameCommands in apply_command:
   ```rust
   GameCommand::QuickSave => {
       let save = build_save_file(&self.game_world, &self.campaign_state, &self.ai_system);
       let save_dir = save_directory();
       match save::quicksave(&save, &save_dir) {
           Ok(()) => log::info!("Quicksave complete"),
           Err(e) => log::error!("Quicksave failed: {}", e),
       }
   }
   GameCommand::QuickLoad => {
       let save_dir = save_directory();
       match save::quickload(&save_dir) {
           Ok(save) => {
               restore_from_save(&mut self.game_world, &mut self.campaign_state, &mut self.ai_system, &save);
               log::info!("Quickload complete");
           }
           Err(e) => log::error!("Quickload failed: {}", e),
       }
   }
   ```

5. Add helper functions:
   ```rust
   fn build_save_file(world: &GameWorld, campaign: &CampaignState, ai: &Option<AiSystem>) -> SaveFile {
       SaveFile {
           version: save::SAVE_VERSION,
           level_num: campaign.current_level,
           game_tick: world.game_tick,
           tick_counter: world.tick_counter,
           game_speed: world.game_speed,
           player_tribe: world.player_tribe,
           ai_update_mult: world.ai_update_mult,
           flags: world.flags,
           rng: world.rng.clone(),
           tribes: world.tribes.clone(),
           ai_script_variables: Vec::new(), // TODO: extract from AiSystem
           ai_every_counters: Vec::new(),   // TODO: extract from AiSystem
       }
   }

   fn save_directory() -> std::path::PathBuf {
       dirs::data_dir().unwrap_or_else(|| std::path::PathBuf::from("."))
           .join("pop3")
           .join("saves")
   }

   fn restore_from_save(
       world: &mut GameWorld,
       campaign: &mut CampaignState,
       ai: &mut Option<AiSystem>,
       save: &SaveFile,
   ) {
       // Restore source-of-truth state
       world.game_tick = save.game_tick;
       world.tick_counter = save.tick_counter;
       world.game_speed = save.game_speed;
       world.player_tribe = save.player_tribe;
       world.ai_update_mult = save.ai_update_mult;
       world.flags = save.flags;
       world.rng = save.rng.clone();
       world.tribes = save.tribes.clone();
       campaign.set_level(save.level_num);

       // Rebuild derived state -- same functions used in the level-load path.
       // These MUST be called to avoid stale terrain/spatial/pathfinding data.
       // Refer to the Loading -> InGame transition in app.rs for the canonical list.
       world.rebuild_terrain_mesh();      // regenerate GPU terrain mesh from heightmap
       world.rebuild_cell_grid();         // repopulate spatial grid from object positions
       world.rebuild_walkability();       // recalculate walkability from terrain + buildings
       // If pathfinding cache exists, invalidate it:
       world.invalidate_pathfinding_cache();

       // Reload AI scripts for the saved level
       if let Some(ref mut ai_sys) = ai {
           let scripts_dir = save_directory().parent()
               .unwrap_or(&std::path::PathBuf::from("."))
               .join("scripts");
           ai_sys.load_level_scripts(save.level_num, &scripts_dir, save.player_tribe);
           // TODO: restore AI EVERY counters and script variables from save
       }
   }
   ```

   NOTE: The rebuild functions (rebuild_terrain_mesh, rebuild_cell_grid, rebuild_walkability, invalidate_pathfinding_cache) must match whatever is called in the Loading -> InGame transition. Read that code path in app.rs during implementation to get the exact function names and call order. The names above are illustrative -- use the actual function names from the codebase.
